192!> Builds a z* coordinate with a minimum thickness in one column. This is the column code
193!! shared by build_zstar_column and build_zstar_columns.
19446370pure subroutine zstar_column(nk, coordinateResolution, CS_min_thickness, depth, total_thickness, &
19546370 z0_top, eta, z_scale, new_zstar_def, zInterface)
196 !$omp declare target
197 integer, intent(in) :: nk !< Number of levels to be generated
198 real, dimension(nk), intent(in) :: coordinateResolution !< Target coordinate resolution, usually in [Z ~> m]

200 !! as depth [Z ~> m] or [H ~> m or kg m-2]
201 real, intent(in) :: depth !< Depth of ocean bottom (positive downward in the
202 !! output units), units may be [Z ~> m] or [H ~> m or kg m-2]
203 real, intent(in) :: total_thickness !< Column thickness (positive definite in the same
204 !! units as depth) [Z ~> m] or [H ~> m or kg m-2]
205 real, intent(in) :: z0_top !< The height of a rigid top, or 0 (positive upward in the same
206 !! units as depth) [Z ~> m] or [H ~> m or kg m-2]
207 real, intent(in) :: eta !< Free surface height (in the same
208 !! units as depth) [Z ~> m] or [H ~> m or kg m-2]
209 real, intent(in) :: z_scale !< Scaling factor from the target coordinate resolution
210 !! in Z to desired units for zInterface, perhaps Z_to_H,
211 !! often [nondim] or [H Z-1 ~> 1 or kg m-3]
212 logical, intent(in) :: new_zstar_def !< If true, use the definition with a rigid top
213 real, dimension(nk+1), intent(inout) :: zInterface !< Absolute positions of interfaces (in the same
214 !! units as depth) [Z ~> m] or [H ~> m or kg m-2]
215 ! Local variables
216 real :: stretching ! A stretching factor for the coordinate [nondim]
217 real :: dh, min_thickness, z_star ! Thicknesses or heights [Z ~> m] or [H ~> m or kg m-2]
218 integer :: k
219
22046370 min_thickness = min( CS_min_thickness, total_thickness/real(nk) )
221
222 ! Conventional z* coordinate:
223 ! z* = (z-eta) / stretching where stretching = (H+eta)/H
224 ! z = eta + stretching * z*
225 ! The above gives z*(z=eta) = 0, z*(z=-H) = -H.
226 ! With a rigid top boundary at eta = z0_top then
227 ! z* = z0 + (z-eta) / stretching where stretching = (H+eta)/(H+z0)
228 ! z = eta + stretching * (z*-z0) * stretching
22946370 stretching = total_thickness / ( depth + z0_top )
230
23146370 if (new_zstar_def) then
232 ! z_star is the notional z* coordinate in absence of upper/lower topography
2330 z_star = 0. ! z*=0 at the free-surface
2340 zInterface(1) = eta ! The actual position of the top of the column
2350 do k = 2,nk
2360 z_star = z_star - coordinateResolution(k-1)*z_scale
237 ! This ensures that z is below a rigid upper surface (ice shelf bottom)
2380 zInterface(k) = min( eta + stretching * ( z_star - z0_top ), z0_top )
239 ! This ensures that the layer in inflated
2400 zInterface(k) = min( zInterface(k), zInterface(k-1) - min_thickness )
241 ! This ensures that z is above or at the topography
2420 zInterface(k) = max( zInterface(k), -depth + real(nk+1-k) * min_thickness )
243 enddo
2440 zInterface(nk+1) = -depth
245
246 else
247 ! Integrate down from the top for a notional new grid, ignoring topography
248 ! The starting position is offset by z0_top which, if z0_top<0, will place
249 ! interfaces above the rigid boundary.
25046370 zInterface(1) = eta
2511576580 do k = 1,nk
2521530210 dh = stretching * coordinateResolution(k)*z_scale ! Notional grid spacing
2531576580 zInterface(k+1) = zInterface(k) - dh
254 enddo
255
256 ! Integrating up from the bottom adjusting interface position to accommodate
257 ! inflating layers without disturbing the interface above

2591576580 do k = nk,1,-1
2601576580 if ( zInterface(k) < (zInterface(k+1) + min_thickness) ) then
261352458 zInterface(k) = zInterface(k+1) + min_thickness
262 endif
263 enddo
264 endif
