131122956pure function log_table(xb, exponent_adjustment) result(a)
132 integer(kind=int_rsize), intent(in) :: xb
133 !< Bit representation of the positive normal input
134 integer(kind=int_rsize), intent(in) :: exponent_adjustment
135 !< Subnormal exponent correction; zero for normal inputs
136 real :: a
137 !< Natural logarithm of the original input [nondim]
138
139 integer, parameter :: binshift = expbit - binbits
140 !< Shift to move fraction subdivision index (idiv) to lowest bits
141 integer(kind=int_rsize) :: K, mb, frac
142 !< Binary exponent, normalized significand bits, and fraction bits
143 integer :: idiv
144 !< Lookup table subdivision index
145 real :: z, r
146 !< Reduced significand and reduced log1p argument [nondim]
147 real :: w, hi, lo
148 !< Compensated partial sums for log(x) [nondim]
149
150 ! For encoding x = 2**K z, z ∈ [1,2),
151 ! log(x) = K ln2 + log(z)
152
153 ! Using z = 1 + r permits a Taylor-like Remez approximation of log(1 + r).
154 ! This can be improved by decomposing [1,2) over N subintervals:
155 ! z ∈ [ 1 + idiv/ndiv, 1 + (idiv + 1)/ndiv )
156
157 ! Extract the fractional part of x, and use its top bits to determine idiv.
158122956 frac = iand(xb, expstep - 1)
159122956 idiv = int(shiftr(frac, binshift))
160
161 ! Extract the exponent and significand from xb
162122956 K = shiftr(xb, expbit) - expbias
163122956 mb = ior(frac, transfer(1., int_mold))
164122956 z = transfer(mb, real_mold)
165
166 ! Apply the subnormal rescaling if present.
167122956 K = K + exponent_adjustment
168
169 ! Use idiv to rescale z to z' = z/c = 1 + r so that 1 + r is close to 1.
170 ! After scaling,
171 ! log(x) = K ln2 + log(c) + log(1 + r).
172
173 ! NOTE: See log data table module for c(idiv) derivation.
174
175 ! Compute r as ((z - c_hi) - c_lo) * (1/c) to avoid precision loss near c.
176122956 r = ((z - c_hi(idiv)) - c_lo(idiv)) * inv_c(idiv)
177
178 ! NOTE: The following can be more accurate, faster, and does not need c_hi
179 ! and c_lo lookups, but must be evaluated by FMA, which cannot be ensured.
180 ! r = z * inv_c(idiv) - 1
181
182 ! Reconstruct log(x) with compensated high and low parts.
183
184 ! Compute the upper part of log(x)
185122956 w = real(K) * ln2_hi + log_c(idiv)
186122956 hi = w + r
187
188 ! XXX: Prevent (w - (w + r)) ~> r optimization by adding a subroundoff term.
189122956 hi = hi + epsilon(real_mold) / 256.
190
191 ! Recover the rounding residual of hi, and append K ln2_lo.
192 ! Cancel w first, since it is the larger of w + r.
193122956 lo = ((w - hi) + r) + real(K) * ln2_lo
194
195 ! Apply the complete residual to hi to recover log(x).
196122956 a = hi + (lo + log1p_tail_remez_4(r))
197122956end function log_table
