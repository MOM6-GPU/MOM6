334 Te_aa, Te_bb, Se_aa, Se_bb, d_A, d_B)
335 real, intent(in) :: dRho_dT !< The derivative of in situ density with temperature [R C-1 ~> kg m-3 degC-1]
336 real, intent(in) :: dRho_dS !< The derivative of in situ density with salinity [R S-1 ~> kg m-3 ppt-1]
337 real, intent(in) :: h_a !< The thickness of the layer above [H ~> m or kg m-2]
338 real, intent(in) :: h_b !< The thickness of the layer below [H ~> m or kg m-2]
339 real, intent(in) :: mix_A !< The time integrated mixing rate of the interface above [H ~> m or kg m-2]
340 real, intent(in) :: mix_B !< The time integrated mixing rate of the interface below [H ~> m or kg m-2]
341 real, intent(in) :: T_a !< The initial temperature of the layer above [C ~> degC]
342 real, intent(in) :: T_b !< The initial temperature of the layer below [C ~> degC]
343 real, intent(in) :: S_a !< The initial salinity of the layer below [S ~> ppt]
344 real, intent(in) :: S_b !< The initial salinity of the layer below [S ~> ppt]
345 real, intent(in) :: Te_aa !< The estimated temperature two layers above rescaled by d_A [C ~> degC]
346 real, intent(in) :: Te_bb !< The estimated temperature two layers below rescaled by d_B [C ~> degC]
347 real, intent(in) :: Se_aa !< The estimated salinity two layers above rescaled by d_A [S ~> ppt]
348 real, intent(in) :: Se_bb !< The estimated salinity two layers below rescaled by d_B [S ~> ppt]
349 real, intent(in) :: d_A !< The rescaling dependency across the interface above [nondim]
350 real, intent(in) :: d_B !< The rescaling dependency across the interface below [nondim]
351 logical :: is_unstable !< The return value, true if the profile is statically unstable
352 !! around the interface in question.
353
354 ! These expressions for the local stability are long, but they have been carefully
355 ! grouped for accuracy even when the mixing rates are huge or tiny, and common
356 ! positive definite factors that would appear in the final expression for the
357 ! locally referenced potential density difference across an interface have been omitted.
358 is_unstable = (dRho_dT * ((h_a * h_b * (T_b - T_a) + &
359 mix_A*mix_B * (d_A*Te_bb - d_B*Te_aa)) + &
360 (h_a*mix_B * (Te_bb - d_B*T_a) + &
361 h_b*mix_A * (d_A*T_b - Te_aa)) ) + &
362 dRho_dS * ((h_a * h_b * (S_b - S_a) + &
